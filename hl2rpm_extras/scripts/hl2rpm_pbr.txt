// HL2RPM: what single materials are made of (physically based materials of the deferred renderer).
//
// Every opaque material gets a roughness, a metalness and a specular level by itself - from its
// $surfaceprop, its folder and its name (r_deferred_pbr_list <text> prints what a material got
// while it loads). The lines below correct that guess where it is wrong or too coarse:
//
//		"<text in the material's name>"		"<roughness> <metalness> <specular>"
//
// roughness	0 (a mirror) .. 1 (matte)
// metalness	0 .. 1 (stock "metal" is painted and rusty: 0.3 - 0.75)
// specular		the mirror reflection of a non-metal, 1 = the usual 4 %
// -1			leave that value to the guess
//
// The first line that fits a material wins. A material that sets $pbr_roughness, $pbr_metalness or
// $pbr_specular in its own .vmt keeps its own. After a change: r_deferred_pbr_reload in the console.
"hl2rpm_pbr"
{
	// polished stone and glazed surfaces (their $surfaceprop says "concrete" or "tile")
	"marble"				"0.22 0 1.3"
	"tileroof"				"0.60 0 1.0"
	"tilebreak"				"0.50 0 1.1"

	// floors people walk on are worn smoother than walls
	"woodfloor"				"0.50 0 1.1"
	"concretefloor"			"0.80 0 0.85"
	"cobble"				"0.78 0 0.85"
	"brickfloor"			"0.82 0 0.8"
	"stonefloor"			"0.78 0 0.85"

	// walls: raw, matte
	"plasterwall"			"0.90 0 0.7"
	"plasterceiling"		"0.92 0 0.7"
	"brickwall"				"0.88 0 0.75"
	"stonewall"				"0.86 0 0.8"

	// metals

	"metalroof"				"0.55 0.40 1.0"
	"metalpipe"				"0.40 0.65 1.0"
	"metaldoor"				"0.48 0.45 1.0"
}
